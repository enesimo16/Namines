using System;
using System.Collections.Generic;
using System.Threading.Tasks;
using Microsoft.AspNetCore.Mvc;
using Microsoft.EntityFrameworkCore;
using NaminesProject.Infrastructure.Data;
using NaminesProject.Domain.Entities;

namespace NaminesProject.API.Controllers;

[ApiController]
[Route("api/[controller]")]
public class OrderItemsController : ControllerBase
{
    private readonly AppDbContext _context;

    public OrderItemsController(AppDbContext context)
    {
        _context = context;
    }

    [HttpGet]
    public async Task<ActionResult<IEnumerable<OrderItems>>> GetAll()
    {
        return await _context.OrderItemses.ToListAsync();
    }

    [HttpGet("{id}")]
    public async Task<ActionResult<OrderItems>> GetById(int id)
    {
        var entity = await _context.OrderItemses.FindAsync(id);
        if (entity == null) return NotFound();
        return entity;
    }

    [HttpPost]
    public async Task<ActionResult<OrderItems>> Create([FromBody] OrderItems entity)
    {
        _context.OrderItemses.Add(entity);
        await _context.SaveChangesAsync();
        return CreatedAtAction(nameof(GetById), new { id = entity.Id }, entity);
    }

    [HttpPut("{id}")]
    public async Task<IActionResult> Update(int id, [FromBody] OrderItems entity)
    {
        if (id != entity.Id) return BadRequest("ID mismatch");

        _context.Entry(entity).State = EntityState.Modified;

        try
        {
            await _context.SaveChangesAsync();
        }
        catch (DbUpdateConcurrencyException)
        {
            if (!await EntityExists(id)) return NotFound();
            throw;
        }

        return NoContent();
    }

    [HttpDelete("{id}")]
    public async Task<IActionResult> Delete(int id)
    {
        var entity = await _context.OrderItemses.FindAsync(id);
        if (entity == null) return NotFound();

        _context.OrderItemses.Remove(entity);
        await _context.SaveChangesAsync();
        return NoContent();
    }

    private Task<bool> EntityExists(int id)
    {
        return _context.OrderItemses.AnyAsync(e => e.Id == id);
    }
}

=== backend/NaminesProject.API/Program.cs ===
using Microsoft.EntityFrameworkCore;
using NaminesProject.Infrastructure.Data;

var builder = WebApplication.CreateBuilder(args);

// Add services to the container with JSON IgnoreCycles configuration.
builder.Services.AddControllers()
    .AddJsonOptions(options =>
    {
        options.JsonSerializerOptions.ReferenceHandler = System.Text.Json.Serialization.ReferenceHandler.IgnoreCycles;
    });

builder.Services.AddEndpointsApiExplorer();
builder.Services.AddSwaggerGen();

// Configure EF Core SQLite database
builder.Services.AddDbContext<AppDbContext>(options =>
    options.UseSqlite(builder.Configuration.GetConnectionString("DefaultConnection")));

// Configure CORS
builder.Services.AddCors(options =>
{
    options.AddPolicy("AllowAll", policy =>
    {
        policy.AllowAnyOrigin()
              .AllowAnyHeader()
              .AllowAnyMethod();
    });
});

var app = builder.Build();

// Automatically run EF migrations
using (var scope = app.Services.CreateScope())
{
    var context = scope.ServiceProvider.GetRequiredService<AppDbContext>();
    context.Database.EnsureCreated();
}

// Configure the HTTP request pipeline.
if (app.Environment.IsDevelopment() || true)
{
    app.UseSwagger();
    app.UseSwaggerUI();
}

app.UseCors("AllowAll");
app.UseAuthorization();
app.MapControllers();

app.Run();

=== backend/NaminesProject.Infrastructure/Data/AppDbContext.cs ===
using System;
using Microsoft.EntityFrameworkCore;
using NaminesProject.Domain.Entities;

namespace NaminesProject.Infrastructure.Data;

public class AppDbContext : DbContext
{
    public AppDbContext(DbContextOptions<AppDbContext> options) : base(options)
    {
    }

    public DbSet<Users> Userses { get; set; }
    public DbSet<Products> Productses { get; set; }
    public DbSet<Orders> Orderses { get; set; }
    public DbSet<OrderItems> OrderItemses { get; set; }

    protected override void OnModelCreating(ModelBuilder modelBuilder)
    {
        base.OnModelCreating(modelBuilder);

        modelBuilder.Entity<Users>().HasKey(e => e.Id);
        modelBuilder.Entity<Products>().HasKey(e => e.Id);
        modelBuilder.Entity<Orders>().HasKey(e => e.Id);
        modelBuilder.Entity<OrderItems>().HasKey(e => e.Id);

        modelBuilder.Entity<Orders>()
            .HasOne(e => e.Users)
            .WithMany(e => e.Orderses)
            .HasForeignKey(e => e.UserId);

        modelBuilder.Entity<OrderItems>()
            .HasOne(e => e.Orders)
            .WithMany(e => e.OrderItemses)
            .HasForeignKey(e => e.OrderId);

        modelBuilder.Entity<OrderItems>()
            .HasOne(e => e.Products)
            .WithMany(e => e.OrderItemses)
            .HasForeignKey(e => e.ProductId);
    }
}

import { useQuery, useMutation, useQueryClient } from '@tanstack/react-query';
import axios from 'axios';
import * as T from './types';

const API_BASE = '/api';

// ─── Users Hooks ──────────────────────────────────────────

export function useGetUserss() {
  return useQuery<T.Users[]>({
    queryKey: ['Userss'],
    queryFn: async () => {
      const res = await axios.get(`${API_BASE}/Users`);
      return res.data;
    }
  });
}

export function useGetUsers(id: any) {
  return useQuery<T.Users>({
    queryKey: ['Users', id],
    queryFn: async () => {
      const res = await axios.get(`${API_BASE}/Users/${id}`);
      return res.data;
    },
    enabled: id !== undefined && id !== null
  });
}

export function useCreateUsers() {
  const queryClient = useQueryClient();
  return useMutation<T.Users, Error, T.Users>({
    mutationFn: async (data) => {
      const res = await axios.post(`${API_BASE}/Users`, data);
      return res.data;
    },
    onSuccess: () => {
      queryClient.invalidateQueries({ queryKey: ['Userss'] });
    }
  });
}

export function useUpdateUsers(id: any) {
  const queryClient = useQueryClient();
  return useMutation<void, Error, T.Users>({
    mutationFn: async (data) => {
      await axios.put(`${API_BASE}/Users/${id}`, data);
    },
    onSuccess: () => {
      queryClient.invalidateQueries({ queryKey: ['Userss'] });
      queryClient.invalidateQueries({ queryKey: ['Users', id] });
    }
  });
}

export function useDeleteUsers() {
  const queryClient = useQueryClient();
  return useMutation<void, Error, any>({
    mutationFn: async (id) => {
      await axios.delete(`${API_BASE}/Users/${id}`);
    },
    onSuccess: () => {
      queryClient.invalidateQueries({ queryKey: ['Userss'] });
